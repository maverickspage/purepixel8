"""
Local 4K/8K Photo + Video Upscaler with AI Colorization
-------------------------------------------------------
pip install flask pillow opencv-python-headless numpy imageio imageio-ffmpeg
Optional AI colorization:  pip install opencv-contrib-python-headless
Model (Zhang et al. 2016): place colorization_release_v2.caffemodel,
  colorization_deploy_v2.prototxt, pts_in_hull.npy in ./models/
  Download from: https://github.com/richzhang/colorization

import io, os, tempfile, subprocess
import cv2
import numpy as np
from flask import Flask, request, send_file, jsonify
from PIL import Image

app = Flask(__name__)
app.config["MAX_CONTENT_LENGTH"] = 2 * 1024 * 1024 * 1024  # 2 GB
TIER_SPECS = {"4k": {"long_edge": 3840, "w": 3840, "h": 2160},
              "8k": {"long_edge": 7680, "w": 7680, "h": 4320}}

# ---- Optional AI colorizer (Zhang et al. 2016) ----
_colornet = None
def _load_colornet():
    global _colornet
    if _colornet is not None:
        return _colornet
    try:
        proto  = "models/colorization_deploy_v2.prototxt"
        model  = "models/colorization_release_v2.caffemodel"
        hull   = "models/pts_in_hull.npy"
        if not (os.path.exists(proto) and os.path.exists(model) and os.path.exists(hull)):
            return None
        net = cv2.dnn.readNetFromCaffe(proto, model)
        pts = np.load(hull).transpose().reshape(2, 313, 1, 1)
        net.getLayer(net.getLayerId("class8_ab")).blobs = [pts.astype(np.float32)]
        net.getLayer(net.getLayerId("conv8_313_rh")).blobs = [np.full([1, 313], 2.606, np.float32)]
        _colornet = net
        return net
    except Exception as e:
        print("Colorizer unavailable:", e)
        return None

def ai_colorize(bgr: np.ndarray) -> np.ndarray:
    """True AI colorization. Returns original if model not loaded."""
    net = _load_colornet()
    if net is None:
        return bgr
    h, w = bgr.shape[:2]
    scale = min(1.0, 512.0 / max(h, w))
    if scale < 1.0:
        small = cv2.resize(bgr, (int(w*scale), int(h*scale)), interpolation=cv2.INTER_AREA)
    else:
        small = bgr
    lab = cv2.cvtColor(small.astype(np.float32) / 255.0, cv2.COLOR_BGR2LAB)
    L = lab[:, :, 0] - 50.0
    net.setInput(cv2.dnn.blobFromImage(L))
    ab = net.forward()[0].transpose((1, 2, 0))
    ab = cv2.resize(ab, (small.shape[1], small.shape[0]))
    L_full = cv2.cvtColor(small.astype(np.float32) / 255.0, cv2.COLOR_BGR2LAB)[:, :, 0]
    lab_out = np.concatenate((L_full[:, :, None], ab), axis=2)
    bgr_small = (cv2.cvtColor(lab_out, cv2.COLOR_LAB2BGR) * 255.0).clip(0, 255).astype(np.uint8)
    return cv2.resize(bgr_small, (w, h), interpolation=cv2.INTER_CUBIC)

# ---- Classical SR pipeline (identical to browser) ----
def neutralize_cast_and_denoise(bgr, denoise_strength, fix_color):
    out = bgr.copy()
    if denoise_strength > 0.02:
        sc = int(12 + denoise_strength * 42); ss = int(3 + denoise_strength * 5)
        out = cv2.bilateralFilter(out, d=5, sigmaColor=sc, sigmaSpace=ss)
    if fix_color:
        result = np.empty_like(out)
        for c in range(3):
            lo, hi = np.percentile(out[:, :, c], (1.5, 98.5))
            if hi - lo > 25:
                st = np.clip((out[:, :, c].astype(np.float32) - lo) * (255.0 / (hi - lo)), 0, 255)
                result[:, :, c] = (0.4 * out[:, :, c] + 0.6 * st).astype(np.uint8)
            else:
                result[:, :, c] = out[:, :, c]
        out = result
    return out

def edge_directed_smoothing(bgr):
    gray = cv2.cvtColor(bgr, cv2.COLOR_BGR2GRAY).astype(np.float32)
    gx = cv2.Sobel(gray, cv2.CV_32F, 1, 0, ksize=3)
    gy = cv2.Sobel(gray, cv2.CV_32F, 0, 1, ksize=3)
    mag = np.hypot(gx, gy); mask = (mag > 28.0)[:, :, None]
    smoothed = cv2.edgePreservingFilter(bgr, flags=1, sigma_s=18, sigma_r=0.22)
    return np.where(mask, (0.45 * bgr + 0.55 * smoothed).astype(np.uint8), bgr)

def synthesize_ibp(hr_bgr, ref_bgr, clarity, synth_gain, iterations=2):
    src_h, src_w = ref_bgr.shape[:2]; hr_h, hr_w = hr_bgr.shape[:2]
    est = hr_bgr.astype(np.float32); ref = ref_bgr.astype(np.float32)
    gain = 0.35 + clarity * 0.65
    for _ in range(iterations):
        sim = cv2.resize(est, (src_w, src_h), interpolation=cv2.INTER_AREA)
        res = cv2.resize(ref - sim, (hr_w, hr_h), interpolation=cv2.INTER_LANCZOS4)
        est = np.clip(est + gain * res, 0, 255)
    if clarity <= 0.01: return est.astype(np.uint8)
    u8 = est.astype(np.uint8)
    b1 = cv2.GaussianBlur(u8, (0, 0), 1.0).astype(np.float32)
    b2 = cv2.GaussianBlur(u8, (0, 0), 2.2).astype(np.float32)
    f = u8.astype(np.float32)
    hi = (f - b1) * (0.65 * clarity * synth_gain)
    mid = (b1 - b2) * (0.35 * clarity * synth_gain)
    edge = np.clip(np.abs(f - b2) / 18.0, 0.15, 1.25)
    sharp = f + (hi + mid) * edge
    k = np.ones((3, 3), np.uint8)
    lo = cv2.erode(u8, k).astype(np.float32); hi_ = cv2.dilate(u8, k).astype(np.float32)
    return np.clip(np.clip(sharp, lo - 6.0, hi_ + 6.0), 0, 255).astype(np.uint8)

def target_size(w, h, tier, mode):
    spec = TIER_SPECS[tier]
    if mode == "cover": return spec["w"], spec["h"]
    s = spec["long_edge"] / max(w, h)
    return max(1, round(w * s)), max(1, round(h * s))

def upscale_frame(bgr, tier="8k", mode="fit", algo="edi_ibp", fix_color=True,
                  denoise=0.40, clarity=0.75, ibp_passes=2, synth_gain=1.0):
    if mode == "cover":
        h, w = bgr.shape[:2]; tr = 16.0 / 9.0
        if (w / h) > tr:
            nw = int(h * tr); bgr = bgr[:, (w - nw) // 2:(w - nw) // 2 + nw]
        elif (w / h) < tr:
            nh = int(w / tr); bgr = bgr[(h - nh) // 2:(h - nh) // 2 + nh, :]
    clean = neutralize_cast_and_denoise(bgr, denoise, fix_color)
    interp = cv2.INTER_CUBIC if algo == "bicubic" else cv2.INTER_LANCZOS4
    if tier == "8k":
        w4, h4 = target_size(clean.shape[1], clean.shape[0], "4k", mode)
        s4 = cv2.resize(clean, (w4, h4), interpolation=interp)
        if algo == "edi_ibp": s4 = edge_directed_smoothing(s4)
        s4 = synthesize_ibp(s4, clean, clarity * 0.85, synth_gain * 0.85, ibp_passes)
        w8, h8 = target_size(clean.shape[1], clean.shape[0], "8k", mode)
        s8 = cv2.resize(s4, (w8, h8), interpolation=interp)
        return synthesize_ibp(s8, s4, clarity, synth_gain, max(1, ibp_passes - 1))
    w4, h4 = target_size(clean.shape[1], clean.shape[0], "4k", mode)
    s4 = cv2.resize(clean, (w4, h4), interpolation=interp)
    if algo == "edi_ibp": s4 = edge_directed_smoothing(s4)
    return synthesize_ibp(s4, clean, clarity, synth_gain, ibp_passes)

# ---- Routes ----
@app.route("/api/upscale-uhd", methods=["POST"])
def upscale_uhd():
    if "image" not in request.files: return jsonify({"error": "No image"}), 400
    file = request.files["image"]
    tier       = request.form.get("target_res", "8k").lower()
    algo       = request.form.get("algorithm", "edi_ibp").lower()
    mode       = request.form.get("fit_mode", "fit").lower()
    fix_color  = request.form.get("restore", "true").lower() == "true"
    denoise    = float(request.form.get("denoise", 0.40))
    clarity    = float(request.form.get("strength", 0.75))
    passes     = int(request.form.get("ibp_passes", 2))
    profile    = request.form.get("synthesis_profile", "crisp").lower()
    colorize   = request.form.get("colorize", "false").lower() == "true"
    gain = {"natural": 0.70, "crisp": 1.0, "maximum": 1.35}.get(profile, 1.0)
    pil = Image.open(file.stream).convert("RGB")
    bgr = cv2.cvtColor(np.array(pil), cv2.COLOR_RGB2BGR)
    out = upscale_frame(bgr, tier, mode, algo, fix_color, denoise, clarity, passes, gain)
    if colorize: out = ai_colorize(out)
    buf = io.BytesIO()
    Image.fromarray(cv2.cvtColor(out, cv2.COLOR_BGR2RGB)).save(buf, "PNG", optimize=True)
    buf.seek(0)
    return send_file(buf, mimetype="image/png", download_name=f"upscaled_{tier}.png")

@app.route("/api/upscale-video", methods=["POST"])
def upscale_video():
    """
    Accepts a video, processes frame-by-frame with the SR pipeline,
    optionally AI-colorizes, writes to MP4 (H.264) via ffmpeg.
    """
    if "video" not in request.files: return jsonify({"error": "No video"}), 400
    file       = request.files["video"]
    tier       = request.form.get("target_res", "4k").lower()
    algo       = request.form.get("algorithm", "edi_ibp").lower()
    fix_color  = request.form.get("restore", "true").lower() == "true"
    denoise    = float(request.form.get("denoise", 0.40))
    clarity    = float(request.form.get("strength", 0.75))
    passes     = int(request.form.get("ibp_passes", 2))
    max_dur    = float(request.form.get("max_duration", 10))
    skip       = max(1, int(request.form.get("frame_skip", 1)))
    colorize   = request.form.get("colorize", "false").lower() == "true"
    gain       = {"natural": 0.70, "crisp": 1.0, "maximum": 1.35}.get(
                     request.form.get("synthesis_profile", "crisp").lower(), 1.0)

    tmp_in  = tempfile.NamedTemporaryFile(suffix=".mp4", delete=False)
    file.save(tmp_in.name); tmp_in.close()
    tmp_out = tempfile.NamedTemporaryFile(suffix=".mp4", delete=False); tmp_out.close()

    cap = cv2.VideoCapture(tmp_in.name)
    fps = cap.get(cv2.CAP_PROP_FPS) or 25.0
    src_w = int(cap.get(cv2.CAP_PROP_FRAME_WIDTH)); src_h = int(cap.get(cv2.CAP_PROP_FRAME_HEIGHT))
    out_w, out_h = target_size(src_w, src_h, tier, "fit")

    # ffmpeg H.264 writer (better than cv2.VideoWriter for 8K)
    ffmpeg_cmd = [
        "ffmpeg", "-y", "-f", "rawvideo", "-vcodec", "rawvideo",
        "-pix_fmt", "bgr24", "-s", f"{out_w}x{out_h}", "-r", str(fps / skip),
        "-i", "-", "-an", "-c:v", "libx264", "-preset", "medium",
        "-crf", "18", "-pix_fmt", "yuv420p", tmp_out.name
    ]
    proc = subprocess.Popen(ffmpeg_cmd, stdin=subprocess.PIPE)

    processed = 0
    idx = 0
    max_frames = int(max_dur * fps)
    try:
        while idx < max_frames:
            ok, frame = cap.read()
            if not ok: break
            if idx % skip == 0:
                out = upscale_frame(frame, tier, "fit", algo, fix_color,
                                    denoise, clarity, passes, gain)
                if colorize: out = ai_colorize(out)
                if out.shape[1] != out_w or out.shape[0] != out_h:
                    out = cv2.resize(out, (out_w, out_h))
                proc.stdin.write(out.tobytes())
                processed += 1
            idx += 1
    finally:
        cap.release(); proc.stdin.close(); proc.wait()
        try: os.unlink(tmp_in.name)
        except OSError: pass

    return send_file(tmp_out.name, mimetype="video/mp4",
                     as_attachment=True, download_name=f"upscaled_{tier}.mp4")

if __name__ == "__main__":
    os.makedirs("models", exist_ok=True)
    app.run(host="127.0.0.1", port=5000, debug=False)
